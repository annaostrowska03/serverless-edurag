\section{Project Overview}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{What is EduRAG?}
  \begin{columns}[t]
    \begin{column}{0.56\textwidth}
      \textbf{EduRAG} is a cloud-native, serverless knowledge base that lets students and researchers \textbf{chat with their study materials} using AI.

      \vspace{0.35cm}
      \begin{block}{Core RAG Idea}
        \small
        Upload a textbook PDF\\
        $\Rightarrow$ system indexes it automatically\\
        $\Rightarrow$ ask anything in natural language\\
        $\Rightarrow$ get an answer with \textbf{page citations}
      \end{block}
      \vspace{0.25cm}
      Built on \textbf{Google Cloud Platform} — fully event-driven, PaaS/Serverless.
    \end{column}
    \begin{column}{0.40\textwidth}
      \begin{block}{Key Parameters}
        \small
        \renewcommand{\arraystretch}{1.2}
        \begin{tabular}{@{}ll@{}}
          LLM          & Gemini 1.5 Pro \\
          Embeddings   & text-embedding-004 \\
          Vector dims  & 768 \\
          Chunk size   & 1{,}000 chars \\
          Top-K        & 3 chunks \\
          Max PDF      & 50 MB \\
          History      & last 3 turns \\
        \end{tabular}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────
\begin{frame}{User Roles}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{block}{Student / End User}
        \small
        \begin{itemize}
          \item Registers with username + password (or Google SSO)
          \item Uploads PDF textbooks per \textbf{subject folder}
          \item Selects which documents to query (checkboxes)
          \item Chats with streaming AI answers
          \item Sees page-level source citations
          \item Manages documents (upload, delete)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Administrator (implicit)}
        \small
        \begin{itemize}
          \item No separate admin UI
          \item Infrastructure via \textbf{Terraform IaC}
          \item Access via GCP Console
          \item Firestore rules enforce per-user isolation
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{block}{Auth Model}
        \small
        Firebase Auth — ID token in\\
        \texttt{Authorization: Bearer \{token\}}\\
        verified server-side on every request.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{Architecture}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{System Architecture}
  \begin{figure}
    \centering
    \includegraphics[width=0.90\linewidth]{architecture.png}
    \caption{\small Two decoupled pipelines: async ingestion (red) and real-time inference (blue)}
  \end{figure}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────
\begin{frame}{Pipeline 1 — Asynchronous Ingestion}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{0.3em}
    \item[\textbf{1.}] Frontend requests a \textbf{pre-signed GCS URL} from the API
    \item[\textbf{2.}] PDF uploaded \textbf{directly to GCS} — never passes through backend\\
          Path: \texttt{uploads/\{user\_id\}/\{subject\}/\{filename\}}
    \item[\textbf{3.}] GCS finalize event $\to$ \textbf{Pub/Sub} message
    \item[\textbf{4.}] \textbf{Worker (Cloud Run)} processes the message:
      \begin{itemize}
        \item Downloads PDF, extracts text with PyPDFLoader
        \item Splits into 1{,}000-char chunks (RecursiveCharacterTextSplitter)
        \item Embeds chunks via \textbf{Vertex AI} \texttt{text-embedding-004}
        \item Stores vectors + metadata in \textbf{ChromaDB} (doc\_id, user\_id, subject, page)
        \item Updates Firestore status: \textit{Uploading} $\to$ \textit{Downloading} $\to$ \textit{Chunking} $\to$ \textit{Vectorizing} $\to$ \textbf{Ready}
      \end{itemize}
    \item[\textbf{5.}] Frontend reflects status instantly via real-time \textbf{Firestore listener}
  \end{enumerate}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────
\begin{frame}{Pipeline 2 — Real-Time Inference}
  \small
  \begin{enumerate}
    \setlength{\itemsep}{0.3em}
    \item[\textbf{1.}] User selects documents via checkboxes and submits a question
    \item[\textbf{2.}] Frontend sends \texttt{POST /ask} with \texttt{question}, \texttt{doc\_ids[]}, \texttt{history[]}
    \item[\textbf{3.}] API verifies Firebase ID token; embeds question via \textbf{Vertex AI}
    \item[\textbf{4.}] \textbf{ChromaDB} similarity search filtered by selected \texttt{doc\_ids}\\
          — top-3 most relevant chunks retrieved as context
    \item[\textbf{5.}] Prompt built: \textbf{System} (context) + last 3 conversation turns + new question
    \item[\textbf{6.}] \textbf{Gemini 1.5 Pro} streams tokens via \textbf{Server-Sent Events (SSE)}\\
          — each chunk: \texttt{data: \{"token": "..."\}}; final: \texttt{data: \{"sources": [...], "done": true\}}
    \item[\textbf{7.}] Frontend renders \textbf{markdown} live; appends clickable \textbf{source citations} (filename, page, excerpt)
  \end{enumerate}
\end{frame}

\section{Storage}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{Storage Design}
  \small
  \renewcommand{\arraystretch}{1.5}
  \begin{tabularx}{\textwidth}{@{} l l l X @{}}
    \toprule
    \textbf{Store} & \textbf{Type} & \textbf{Consistency} & \textbf{Role} \\
    \midrule
    \textbf{Cloud Storage} & Blob & Strong & Raw PDFs. Write-once per upload; read-once by Worker. Path encodes user + subject. \\
    \textbf{Firestore} & NoSQL document & Strong & Per-user document metadata \& processing status. Drives real-time UI listener. \\
    \textbf{ChromaDB} & Vector (float32) & Eventual & Text embeddings + chunk metadata. Write-once on ingestion; read-many on every chat query. \\
    \bottomrule
  \end{tabularx}

  \vspace{0.25cm}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{block}{Data Volume Estimate}
        \footnotesize
        10{,}000 users $\times$ 50 docs $\times$ 500 chunks $\times$ 768 floats $\approx$ \textbf{28 GB vectors}\\
        GCS PDFs: up to \textbf{25 TB}
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Document Deletion}
        \footnotesize
        Atomic across all 3 stores:\\
        ChromaDB vectors $\to$ GCS blob $\to$ Firestore doc
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\section{SLA / SLO / SLI}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{SLA / SLO / SLI}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{SLA — commitment to users}
        \small
        \textbf{99.9\% uptime} for the Chat UI and Inference API, backed by GCP managed services SLAs.
      \end{block}
      \vspace{0.2cm}
      \begin{block}{SLOs — objectives}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.2em}
          \item \textbf{Latency:} 95\% of \texttt{/ask} requests — first token in $<\!2\,$s
          \item \textbf{Ingestion:} 99\% of PDFs ($<$50 MB) vectorised within 3 minutes
          \item \textbf{Availability:} error rate $<\!0.1\%$ over any 30-day window
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{SLIs — how we measure}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.2em}
          \item \textbf{Time-to-first-token:} Cloud Run request latency to first SSE chunk (Cloud Monitoring)
          \item \textbf{Ingestion duration:} Firestore timestamp delta \textit{Uploading} $\to$ \textit{Ready}
          \item \textbf{Error rate:} HTTP 5xx / total requests (Cloud Run metrics)
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{alertblock}{ChromaDB caveat}
        \small
        Eventual consistency — newly uploaded chunks may not appear in queries for a few seconds.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Large-Scale Assumptions}
% ═══════════════════════════════════════════════════════════════════════════

\begin{frame}{Large-Scale Assumptions}
  \begin{columns}[t]
    \begin{column}{0.46\textwidth}
      \begin{block}{Load model}
        \small
        \renewcommand{\arraystretch}{1.2}
        \begin{tabular}{@{}ll@{}}
          Registered users    & 10{,}000 \\
          Concurrent users    & 500 \\
          Docs per user       & up to 50 \\
          Doc size            & up to 50 MB \\
          Chat requests       & 1{,}000 QPS \\
          Uploads/hour        & 200 \\
        \end{tabular}
      \end{block}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{block}{Why it scales}
        \small
        \begin{itemize}
          \setlength{\itemsep}{0.2em}
          \item \textbf{Cloud Run} — scales to 0, auto scales out under load
          \item \textbf{Pub/Sub} — buffers upload bursts; worker consumes async
          \item \textbf{Direct-to-GCS upload} — bypasses API entirely for binary transfer
          \item \textbf{Firestore} — handles thousands of concurrent listeners natively
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{alertblock}{Bottleneck}
        \small
        ChromaDB on a \textbf{single VM} — main scalability constraint. Horizontal sharding requires a managed vector DB.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Key Features}

\begin{frame}{Key Features — Infrastructure \& Security}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Infrastructure}
        \begin{itemize}
          \setlength{\itemsep}{0.25em}
          \item Full \textbf{Terraform IaC} — Cloud Run, GCS, Pub/Sub, Firestore, IAM, Artifact Registry
          \item Event-driven: GCS $\to$ Pub/Sub $\to$ Worker
          \item Containerised (Docker + gunicorn)
          \item \textbf{--allow-unauthenticated} Cloud Run + Firebase token auth in Flask
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Security}
        \begin{itemize}
          \setlength{\itemsep}{0.25em}
          \item \textbf{Firebase Auth} — Google SSO and username/password (synthetic email trick: \texttt{name@edurag.app})
          \item Per-user data isolation — Firestore sub-collection + ChromaDB \texttt{user\_id} filter
          \item \textbf{Pre-signed GCS URLs} — PDFs never pass through backend
          \item Token verified server-side via \texttt{firebase-admin} on every request
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────
\begin{frame}{Key Features — User Experience}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Document Management}
        \begin{itemize}
          \setlength{\itemsep}{0.25em}
          \item \textbf{Subject folders} — organise PDFs by topic (like Google Drive)
          \item \textbf{Per-document checkboxes} — choose which docs to query (like NotebookLM)
          \item \textbf{Real-time status} — Downloading $\to$ Chunking $\to$ Vectorizing $\to$ Ready (Firestore listener)
          \item \textbf{Delete document} — removes vectors, PDF, and Firestore record atomically
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Chat Interface}
        \begin{itemize}
          \setlength{\itemsep}{0.25em}
          \item \textbf{Streaming responses} (SSE) — tokens appear as they are generated
          \item \textbf{Markdown rendering} — bold, lists, code blocks, headings
          \item \textbf{Source citations} — filename, page number, text excerpt per answer
          \item \textbf{Multi-turn conversation} — last 3 turns kept as context
          \item \textbf{Copy to clipboard} and \textbf{clear chat}
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
