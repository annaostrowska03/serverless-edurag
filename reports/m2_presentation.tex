\section{Project Overview}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}[t,shrink=3]{What is EduRAG?}
  \begin{columns}[t]
    \begin{column}{0.55\textwidth}
      \textbf{EduRAG} is a cloud-native, serverless knowledge base that lets
      students \textbf{chat with their study materials} using AI.

      \vspace{0.3cm}
      \begin{block}{Core RAG loop}
        Upload a PDF
        $\;\Rightarrow\;$ automatic indexing
        $\;\Rightarrow\;$ ask in natural language
        $\;\Rightarrow\;$ answer with \textbf{page citations}
      \end{block}

      \vspace{0.25cm}
      {\small Built on \textbf{Google Cloud Platform} —
      fully event-driven, PaaS/Serverless.}
    \end{column}
    \begin{column}{0.41\textwidth}
      \begin{block}{Key Parameters}
        {\footnotesize\renewcommand{\arraystretch}{1.2}
        \begin{tabular}{@{}ll@{}}
          LLM           & Gemini 2.5 Flash \\
          Embeddings    & text-embedding-004 \\
          Vector dims   & 768 \\
          Chunk size    & 1{,}000 chars \\
          Chunk overlap & 100 chars \\
          Top-K         & 3 chunks \\
          Max PDF       & 50 MB \\
          History       & last 3 turns \\
        \end{tabular}}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ──────────────────────────────────────────────────────────────────────────
\begin{frame}[t,shrink=3]{User Roles}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{block}{Student / End User}
        \begin{itemize}\setlength\itemsep{0.15em}
          \item Registers with username + password
          \item Uploads PDFs into \textbf{subject folders}
          \item Searches \& filters documents
          \item Selects documents to query (checkboxes)
          \item Chats with streaming AI answers + citations
          \item Manages \textbf{persistent chat sessions}
          \item Deletes documents on demand
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Administrator (implicit)}
        \begin{itemize}\setlength\itemsep{0.15em}
          \item No separate admin UI
          \item Infrastructure via \textbf{Terraform IaC}
          \item Access via GCP Console
          \item Firestore rules enforce isolation
        \end{itemize}
      \end{block}
      \vspace{0.25cm}
      \begin{block}{Auth Model}
        {\small Firebase Auth — ID token\\
        \texttt{Authorization: Bearer \{token\}}\\
        verified server-side on every request.}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

% ──────────────────────────────────────────────────────────────────────────
\begin{frame}[t,shrink=3]{Pipeline 1 — Asynchronous Ingestion}
  \small
  \begin{enumerate}\setlength\itemsep{0.2em}
    \item[\textbf{1.}] Frontend requests a \textbf{pre-signed GCS URL} from the API
    \item[\textbf{2.}] PDF uploaded \textbf{directly to GCS} —
          {\footnotesize\texttt{uploads/\{uid\}/\{subject\}/\{file\}}}
    \item[\textbf{3.}] GCS finalize event $\to$ \textbf{Pub/Sub} message
    \item[\textbf{4.}] \textbf{Worker (Cloud Run)} processes the message:
      \begin{itemize}\setlength\itemsep{0.05em}
        \footnotesize
        \item Downloads PDF; extracts text with \texttt{PyPDFLoader}
        \item Splits into 1{,}000-char chunks (overlap 100)
        \item Embeds chunks via \textbf{Vertex AI} \texttt{text-embedding-004}
        \item Stores vectors + metadata in \textbf{ChromaDB}
        \item Firestore: Uploading $\to$ Downloading $\to$ Chunking
              $\to$ Vectorizing $\to$ \textbf{Ready}
      \end{itemize}
    \item[\textbf{5.}] UI updates status in real-time via \textbf{Firestore listener}
  \end{enumerate}
\end{frame}

% ──────────────────────────────────────────────────────────────────────────
\begin{frame}[t,shrink=3]{Pipeline 2 — Real-Time Inference}
  \small
  \begin{enumerate}\setlength\itemsep{0.2em}
    \item[\textbf{1.}] User selects documents and submits a question
    \item[\textbf{2.}] \texttt{POST /ask}: question, doc\_ids[], history[]
    \item[\textbf{3.}] API verifies Firebase ID token; embeds question via \textbf{Vertex AI}
    \item[\textbf{4.}] \textbf{Smart retrieval:} question names a file
          $\Rightarrow$ only that doc searched;\\
          otherwise all selected docs $\Rightarrow$ top-3 chunks by cosine similarity
    \item[\textbf{5.}] Prompt: System (context) + last 3 turns + question
    \item[\textbf{6.}] \textbf{Gemini 2.5 Flash} streams tokens via \textbf{SSE}\\
          {\footnotesize\texttt{data:\{"token":"..."\}}
          $\;\cdots\;$
          \texttt{data:\{"sources":[...],"done":true\}}}
    \item[\textbf{7.}] UI renders \textbf{markdown} live; shows \textbf{citations};
          session saved to Firestore
  \end{enumerate}
\end{frame}

  \renewcommand{\arraystretch}{1.3}

    \toprule
    \textbf{Store} & \textbf{Consistency} & \textbf{Role} & \textbf{Pattern} \\
    \midrule
    Cloud Storage  & Strong   &
      Raw PDFs. Path: \texttt{uploads/uid/subject/file} &
      Write-once; read-once by Worker \\
    Firestore      & Strong   &
      Doc metadata + status; persistent chat sessions &
      High R/W; real-time listeners \\
    ChromaDB       & Eventual &
      Text embeddings + metadata (doc, user, subject, page) &
      Write-once; read-many on every query \\
    \bottomrule
  \end{tabularx}

  \vspace{0.25cm}
  \begin{columns}[t]
    \begin{column}{0.50\textwidth}
      \begin{block}{Volume Estimate}
        {\footnotesize
        10k users $\times$ 50 docs $\times$ 500 chunks $\times$ 768 floats
        $\approx$ \textbf{28 GB} vectors\\
        GCS PDFs: up to \textbf{25 TB}}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Document Deletion}
        {\footnotesize Atomic across all 3 stores:\\
        ChromaDB $\to$ GCS $\to$ Firestore}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

      \vspace{0.2cm}
      \begin{block}{SLOs — objectives}
        \begin{itemize}\setlength\itemsep{0.2em}
          \item \textbf{Latency:} 95\% of \texttt{/ask} requests —
                first token $<\!2\,$s
          \item \textbf{Ingestion:} 99\% of PDFs ($<$50 MB)
                vectorised $<\!3\,$min
          \item \textbf{Availability:} error rate $<\!0.1\%$
                (30-day window)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{SLIs — how we measure}
        \begin{itemize}\setlength\itemsep{0.2em}
          \item \textbf{Time-to-first-token:}
                Cloud Run latency to first SSE chunk
          \item \textbf{Ingestion duration:}
                Firestore delta Uploading $\to$ Ready
          \item \textbf{Error rate:}
                HTTP 5xx / total (Cloud Run metrics)
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{alertblock}{ChromaDB caveat}
        {\small Eventual consistency — new chunks may not be
        queryable for a few seconds after ingestion.}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Large-Scale Assumptions}
% ════════════════════════════════════════════════════════════════════════════

\begin{frame}[t,shrink=3]{Large-Scale Assumptions}
  \begin{columns}[t]
    \begin{column}{0.44\textwidth}
      \begin{block}{Load model}
        {\small\renewcommand{\arraystretch}{1.2}
        \begin{tabular}{@{}ll@{}}
          Registered users  & 10{,}000 \\
          Concurrent users  & 500 \\
          Docs per user     & up to 50 \\
          Doc size          & up to 50 MB \\
          Chat requests     & 1{,}000 QPS \\
          Uploads/hour      & 200 \\
        \end{tabular}}
      \end{block}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{block}{Why it scales}
        \begin{itemize}\setlength\itemsep{0.2em}
          \item \textbf{Cloud Run} — scale-to-zero,
                auto-scales under load
          \item \textbf{Pub/Sub} — buffers upload bursts;
                Worker consumes async
          \item \textbf{Direct-to-GCS} — binary transfer
                bypasses API entirely
          \item \textbf{Firestore} — thousands of concurrent
                real-time listeners
        \end{itemize}
      \end{block}
      \vspace{0.2cm}
      \begin{alertblock}{Bottleneck}
        {\small ChromaDB on a \textbf{single VM} — sharding
        requires a managed vector DB.}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
